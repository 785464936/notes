```dataviewjs
const pages = dv.pages()
  .where(p => 
    (p.project === "[[复习]]" || (p.tags && p.tags.includes("复习"))) &&
    p.file.folder !== "Templates" &&
    !p.file.folder.startsWith("Templates/")
  );

const unarchived = pages.where(p => !p.isArchived);
const archived = pages.where(p => p.isArchived === true);

//dv.header(3, "📂 未归档");
if (unarchived.length) {
    dv.list(unarchived.file.link);
} else {
    dv.paragraph("无未归档笔记");
}

dv.header(3, "📦 Archived");
if (archived.length) {
    dv.list(archived.file.link);
} else {
    dv.paragraph("无已归档笔记");
}
```
